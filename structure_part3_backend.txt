// Helper function to create the secret hash - FIXED VERSION
const createSecretHash = (username, clientId, clientSecret) => {
  if (!clientSecret) {
    console.log('🔍 No client secret provided, skipping SECRET_HASH');
    return null;
  }
  
  console.log('🔐 Creating SECRET_HASH for user:', username);
  const message = username + clientId;
  const hmac = crypto.createHmac('sha256', clientSecret);
  hmac.update(message);
  const hash = hmac.digest('base64');
  console.log('✅ SECRET_HASH created successfully');
  return hash;
};

const signupUser = async (email, password, name, role) => {
  try {
    console.log('⭐ Creating new user in Cognito:', email);
    
    const client = createCognitoClient();
    
    // Prepare user attributes
    const userAttributes = [
      { Name: 'email', Value: email },
      { Name: 'email_verified', Value: 'true' },
      { Name: 'name', Value: name }
    ];
    
    // Add role if specified
    if (role) {
      userAttributes.push({ Name: 'custom:role', Value: role });
    }
    
    // Create the user
    const command = new AdminCreateUserCommand({
      UserPoolId: process.env.COGNITO_USER_POOL_ID,
      Username: email,
      TemporaryPassword: password,
      UserAttributes: userAttributes,
      MessageAction: 'SUPPRESS' // Don't send a welcome email
    });
    
    const result = await client.send(command);
    
    // Set the password to a permanent one
    if (result.User) {
      try {
        const setPwdCommand = new AdminSetUserPasswordCommand({
          UserPoolId: process.env.COGNITO_USER_POOL_ID,
          Username: email,
          Password: password,
          Permanent: true
        });
        
        await client.send(setPwdCommand);
        console.log('✅ Password set as permanent');
      } catch (pwdError) {
        console.error('❌ Error setting permanent password:', pwdError);
        // Don't fail the whole request if this step fails
      }
    }
    
    console.log('✅ User created successfully in Cognito');
    return {
      success: true,
      user: {
        id: result.User.Username,
        email,
        name,
        role: role || 'Operator'
      }
    };
  } catch (error) {
    console.error('❌ Signup failed:', error);
    return { success: false, error: error.message };
  }

router.post('/', async (req, res) => {
  try {
    const { username, password } = req.body;
    
    if (!username || !password) {
      return res.status(400).json({ 
        error: 'Username and password required',
        code: 'MISSING_CREDENTIALS'
      });
    }
    
    logInfo(`Login attempt for user: ${username}`);
    const authResult = await authenticateUser(username, password);
    
    if (!authResult.success) {
      logInfo(`Login failed for user: ${username} - ${authResult.error}`);
      return res.status(401).json({ 
        error: authResult.error || 'Authentication failed',
        code: 'AUTH_FAILED' 
      });
    }
    
    // Set refresh token in an HTTP-only cookie for security
    res.cookie('refreshToken', authResult.tokens.refreshToken, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      maxAge: 30 * 24 * 60 * 60 * 1000, // 30 days
      sameSite: 'strict'
    });
    
    logInfo(`Login successful for user: ${username}`);
    
    // Return access and ID tokens to the client
    res.json({
      success: true,
      accessToken: authResult.tokens.accessToken,
      idToken: authResult.tokens.idToken,
      refreshToken: authResult.tokens.refreshToken, // Include for clients without cookie support
      expiresIn: authResult.tokens.expiresIn,
      message: 'Login successful'
    });
  } catch (error) {
    logError(`Login error: ${error.message}`);
    res.status(500).json({ 
      error: 'Login failed: ' + (error.message || 'Unknown error'),
      code: 'LOGIN_ERROR'
    });
  }
});

// Enhanced token refresh route with better error handling
router.post('/refresh-token', async (req, res) => {
  try {
    // Try to get refresh token from cookie first, then from request body
    const refreshToken = req.cookies.refreshToken || req.body.refreshToken;
    
    if (!refreshToken) {
      logInfo('Refresh token request without token');
      return res.status(401).json({ 
        error: 'Refresh token required',
        code: 'MISSING_REFRESH_TOKEN'
      });
    }
    
    logInfo('Attempting token refresh');
    const refreshResult = await refreshTokens(refreshToken);
    
    if (!refreshResult.success) {
      logError(`Token refresh failed: ${refreshResult.error}`);
      
      // Clear the refresh token cookie if it's invalid
      res.clearCookie('refreshToken');
      
      return res.status(401).json({ 
        error: refreshResult.error || 'Token refresh failed',
        code: 'REFRESH_FAILED'
      });
    }
    
    logInfo('Token refreshed successfully');
    
    // Return new tokens
    res.json({
      success: true,
      accessToken: refreshResult.tokens.accessToken,
      idToken: refreshResult.tokens.idToken,
      expiresIn: refreshResult.tokens.expiresIn,
      message: 'Tokens refreshed successfully'
    });
  } catch (error) {
    logError(`Token refresh error: ${error.message}`);
    
    // Clear potentially corrupted refresh token
    res.clearCookie('refreshToken');
    
    res.status(500).json({ 
      error: 'Token refresh failed: ' + (error.message || 'Unknown error'),
      code: 'REFRESH_ERROR'
    });
  }
});

async function deleteImagesFromS3(images) {
  if (!images || !Array.isArray(images) || images.length === 0) {
    return;
  }

  const deletePromises = images.map(async (image) => {
    if (image.key) {
      try {
        // Delete original image
        await s3.deleteObject({ Bucket: bucketName, Key: image.key }).promise();
        
        // Delete thumbnail if it exists
        if (image.thumbnailKey) {
          await s3.deleteObject({ Bucket: bucketName, Key: image.thumbnailKey }).promise();
        }
      } catch (deleteError) {
        console.error(`❌ Failed to delete image ${image.key} from S3:`, deleteError);
      }
    }
  });

  await Promise.all(deletePromises);
}

// server/src/mqtt/connection.js
const mqtt = require('mqtt');
const fs = require('fs');
const path = require('path');
const { logDebug, logError, logInfo } = require('../utils/logger');
const { processRfidEvent } = require('./eventProcessing');
const topics = require('./topics');
const { mqttConfig } = require('./config');

// Track the MQTT connection status
let mqttStatus = {
  connected: false,
  lastError: null,
  reconnectAttempts: 0,
  lastReconnectTime: 0
};

// MQTT client instance
let client = null;

// Function to emit the current status to all clients
const broadcastStatus = (io) => {
  if (!io) return;
  
  io.emit('mqtt_status', {
    connected: mqttStatus.connected,
    lastError: mqttStatus.lastError,
    timestamp: new Date().toISOString()
  });
};

// Connect to MQTT broker
const connectToMqtt = (io) => {
  try {
    logInfo(`Connecting to MQTT broker at ${mqttConfig.brokerUrl}`);

    // Only create one client instance
    if (client && client.connected) {
      logInfo('MQTT client already connected, reusing existing client');
      
      // Broadcast current status
      if (io) {
        broadcastStatus(io);
      }
      
      return client;
    }

    const options = {
      clientId: `rfid-server-${Math.random().toString(16).slice(2)}`,
      clean: true,
      reconnectPeriod: 5000,
      connectTimeout: 30000,
      keepalive: 60,
    };

    if (mqttConfig.useTls) {
      try {
        options.cert = fs.readFileSync(path.resolve(mqttConfig.certPath));
        options.key = fs.readFileSync(path.resolve(mqttConfig.keyPath));
        options.ca = fs.readFileSync(path.resolve(mqttConfig.caPath));
        options.rejectUnauthorized = true;
        logDebug('✅ TLS certs loaded successfully');
      } catch (err) {
        logError(`❌ Failed to load TLS certs: ${err.message}`);
        mqttStatus.lastError = `TLS certificate error: ${err.message}`;
        if (io) io.emit('mqtt_error', { message: mqttStatus.lastError });
        return null;
      }
    }

    // Create the client
    client = mqtt.connect(mqttConfig.brokerUrl, options);

    // Set up MQTT client event handlers
    client.on('connect', () => {
      logInfo('✅ Connected to MQTT broker');
      mqttStatus.connected = true;
      mqttStatus.lastError = null;
      mqttStatus.reconnectAttempts = 0;
      mqttStatus.lastReconnectTime = 0;
      
      if (io) broadcastStatus(io);
      
      // Subscribe to topics
      const topicList = topics.getAllTopics();
      topicList.forEach((topic) => {
        client.subscribe(topic, (err) => {
          if (err) logError(`❌ Subscription error: ${err.message}`);
          else logDebug(`✅ Subscribed to ${topic}`);
        });
      });
    });

    client.on('reconnect', () => {
      const now = Date.now();
      
      // Only log if it's been a while since the last attempt
      if (now - mqttStatus.lastReconnectTime > 5000) {
        mqttStatus.reconnectAttempts++;
        mqttStatus.lastReconnectTime = now;
        logDebug(`Attempting to reconnect to MQTT broker (attempt ${mqttStatus.reconnectAttempts})`);
        
        if (io) {
          io.emit('mqtt_status', {
            connected: false,
            reconnecting: true,
            attempts: mqttStatus.reconnectAttempts,
            timestamp: new Date().toISOString()
          });
        }
      }
    });

    client.on('error', (err) => {
      logError(`❌ MQTT client error: ${err.message}`);
      mqttStatus.lastError = err.message;
      
      if (io) {
        io.emit('mqtt_error', { 
          message: err.message,
          timestamp: new Date().toISOString()
        });
      }
    });

    client.on('offline', () => {
      logError('❌ MQTT client offline');
      mqttStatus.connected = false;
      
      if (io) broadcastStatus(io);
    });

    client.on('message', (topic, message) => {
      try {
        const parsed = JSON.parse(message.toString());
        
        // Process the event and emit it to all clients
        if (io) {
          io.emit('rfid_event', parsed);
          processRfidEvent(parsed, io);
        }
      } catch (err) {
        logError(`❌ Message parsing error: ${err.message}`);
      }
    });

    // Set up socket listeners if io is provided
    if (io) {
      io.on('connection', (socket) => {
        // Send current status when a client connects
        socket.emit('mqtt_status', {
          connected: mqttStatus.connected,
          lastError: mqttStatus.lastError,
          timestamp: new Date().toISOString()
        });
        
        // Listen for status requests
        socket.on('get_mqtt_status', () => {
          socket.emit('mqtt_status', {
            connected: mqttStatus.connected,
            lastError: mqttStatus.lastError,
            timestamp: new Date().toISOString()
          });
        });
      });
    }

    return client;
  } catch (err) {
    logError(`❌ MQTT connection error: ${err.message}`);
    mqttStatus.lastError = err.message;
    
    if (io) {
      io.emit('mqtt_error', { 
        message: err.message,
        timestamp: new Date().toISOString()
      });
    }
    
    return null;
  }
};

// Make sure MQTT is connected on server start and stays connected
const ensureMqttConnected = (io) => {
  // Initial connection
  let mqttClient = connectToMqtt(io);
  
  // Set up a check every minute to ensure connection
  setInterval(() => {

      logInfo('MQTT connection check: Not connected, attempting to reconnect');
      mqttClient = connectToMqtt(io);
    } else {
      logDebug('MQTT connection check: Connected');
    }
  }, 60000);
  
  return mqttClient;
};

  }

const setupConnectionHandlers = (client, io) => {

  // MQTT message handler
  client.on('message', async (topic, message) => {
    try {
      const payload = JSON.parse(message.toString());
      
      // Skip heartbeat and system messages - too noisy
      if (payload.type === 'heartbeat' || payload.component === 'RG') {
        return;
      }

      // Only forward actual tag events to frontend
      if (topic.includes('tagdata') || topic.includes('events')) {
        io.emit('rfid_event', payload);
      }

      // Enhanced RFID event processing with off-site detection
      if (payload.tagId) {
        // Find order by tag ID
        const order = await Order.findOne({
          where: { tagId: payload.tagId }
        });

        if (order) {
          // ✅ FIXED: Check order status FIRST, before any processing
          if (order.status === 'completed') {
            logger.debug(`📱 Ignoring RFID scan for completed order: ${order.orderNumber} (${order.tagId})`);
            return; // Exit completely - no database updates, no websocket events
          }

          // Extract reader information from payload
          const tagId = payload.tagId;
          const hostName = payload.data?.hostName?.toLowerCase() || payload.hostName?.toLowerCase();
          const antenna = payload.data?.antenna || payload.antenna || '1';
          
          // Log the scan event
          logger.debug(`📡 RFID scan: Tag ${tagId} at reader ${hostName}, antenna ${antenna}`);
          
          // Resolve reader location to department/station
          let stationInfo = null;
          if (hostName && antenna) {
            const result = resolveDepartmentAndStation(hostName, antenna);
            
            if (result && !result.error && result.department && result.station) {
              stationInfo = {
                departmentId: getDepartmentId(result.department), // Helper function to get dept ID
                departmentName: result.department,
                stationName: result.station
              };
              
              logger.debug(`📍 Resolved location: ${result.department} > ${result.station}`);
            } else {
              logger.debug(`⚠️ Could not resolve location for reader ${hostName}, antenna ${antenna}`);
            }
          }
          
          // 🆕 NEW: Rate limiting for location updates
          const now = Date.now();
          const lastUpdate = lastLocationUpdates.get(tagId);
          const locationKey = `${stationInfo?.departmentName}-${stationInfo?.stationName}`;
          const lastLocationKey = lastLocationUpdates.get(`${tagId}_location`);
          
          // Check if this is a different location and enough time has passed
          const shouldUpdateLocation = !lastUpdate || 
            (now - lastUpdate > SCAN_RATE_LIMIT) || 
            (lastLocationKey !== locationKey);
          
          console.log('🔍 Rate limiting check:', {
            tagId,
            orderNumber: order.orderNumber,
            currentLocation: `${order.currentDepartment} > ${order.currentStation}`,
            newLocation: locationKey,
            lastUpdate: lastUpdate ? new Date(lastUpdate).toISOString() : 'never',
            timeSinceLastUpdate: lastUpdate ? now - lastUpdate : 'N/A',
            shouldUpdate: shouldUpdateLocation,
            rateLimitMs: SCAN_RATE_LIMIT
          });

          if (shouldUpdateLocation && stationInfo) {
            console.log(`📍 Updating location for order ${order.orderNumber}: ${locationKey}`);
            
            // Update the rate limiting tracking
            lastLocationUpdates.set(tagId, now);
            lastLocationUpdates.set(`${tagId}_location`, locationKey);
            
            // ✅ Now safe to update RFID scan (order is confirmed not completed)
            await orderService.updateOrderRfidScan(order.id, stationInfo);

            });

                  stationInfo
                });

            const updatedOrder = await Order.findByPk(order.id);
