/**
 * Get user data from ID token

export const getUser = (): User | null => {
  try {
    const idToken = getIdToken();
    
    if (!idToken) {
      return null;
    }

    const decoded: DecodedToken = jwtDecode(idToken);

    if (!decoded || !decoded.sub) {
      return null;
    }

    // Extract name from various possible fields
    let name = '';
    if (decoded.name) {
      name = decoded.name;
    } else if (decoded.given_name || decoded.family_name) {
      name = `${decoded.given_name || ''} ${decoded.family_name || ''}`.trim();
    } else if (decoded.username) {
      name = decoded.username;
    } else if (decoded.email) {
      name = decoded.email.split('@')[0];
    } else {
      name = decoded.sub;
    }

    const role = decoded['custom:role'] || decoded.customrole || 'User';
    const email = decoded.email || '';

    return {
      id: decoded.sub,
      email: email,
      name: name,
      role: role,
    };
  } catch (error) {
    console.error('Error getting user from ID token:', error);
    return null;
  }

/**
 * Create authenticated fetch wrapper that handles token refresh
 */
export const authenticatedFetch = async (url: string, options: RequestInit = {}): Promise<Response> => {
  const makeRequest = async (token: string): Promise<Response> => {
    return fetch(url, {
      ...options,
      headers: {
        ...options.headers,
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json',
      },
      credentials: 'include',
    });
  };

  // Check for inactivity before making request
  if (isUserInactive()) {
    console.log('❌ User inactive for 12 hours, blocking request and logging out...');
    await logout();
    throw new Error('Session expired due to inactivity');
  }

  // Get current token
  let token = getAuthToken();
  if (!token) {
    throw new Error('No access token available');
  }

  // Make the request
  let response = await makeRequest(token);

  // If unauthorized, try to refresh token and retry once
  if (response.status === 401) {
    console.log('🔄 Received 401, attempting Cognito token refresh...');
    
    const refreshed = await refreshTokens();
    if (refreshed) {
      token = getAuthToken();
      if (token) {
        console.log('🔄 Retrying request with new Cognito token...');
        response = await makeRequest(token);
      }
    }
    
    // If still unauthorized after refresh, logout
    if (response.status === 401) {
      console.log('❌ Still unauthorized after Cognito refresh, logging out...');
      await logout();
      throw new Error('Session expired');
    }
  }

  // Update activity on successful requests
  if (response.ok) {
    updateActivity();
  }

  return response;
};```

import { API_BASE_URL } from './constants';
import { FetchOptions } from './types';
import { handleApiError } from './errorHandler';
import { toast } from 'sonner';
import { authenticatedFetch, updateActivity } from '../../services/authService';

// Cache duration in milliseconds (5 minutes)
const CACHE_DURATION = 5 * 60 * 1000;

interface CacheItem<T> {
  data: T;
  timestamp: number;
}

class ApiCache {
  private cache: Record<string, CacheItem<any>> = {};

  // Get cached data if valid, otherwise return null
  get<T>(key: string): T | null {
    const item = this.cache[key];
    if (!item) return null;

    const now = Date.now();
    if (now - item.timestamp > CACHE_DURATION) {
      // Cache expired
      delete this.cache[key];
      return null;
    }

    return item.data;
  }

  // Store data in cache
  set<T>(key: string, data: T): void {
    this.cache[key] = {
      data,
      timestamp: Date.now(),
    };
  }

  // Clear specific cache entry
  clear(key: string): void {
    delete this.cache[key];
  }

  // Clear all cache
  clearAll(): void {
    this.cache = {};
  }
}

// API cache instance
export const apiCache = new ApiCache();

/**
 * Generic API request handler using enhanced authenticatedFetch
 */
export async function apiRequest<T>(
  endpoint: string,
  options: FetchOptions = {}
): Promise<T> {
  const { requiresAuth = true, useCache = true, ...fetchOptions } = options;
  
  // Only use cache for GET requests
  const canUseCache = useCache && (!fetchOptions.method || fetchOptions.method === 'GET');
  
  // Generate cache key based on endpoint and query params
  const cacheKey = `${endpoint}${fetchOptions.body ? `-${JSON.stringify(fetchOptions.body)}` : ''}`;
  
  // Try to get from cache first for GET requests
  if (canUseCache) {
    const cachedData = apiCache.get<T>(cacheKey);
    if (cachedData) {
      // Update activity when serving from cache
      updateActivity();
      return cachedData;
    }
  }

  // Build full URL
  let url = '';
  if (endpoint.startsWith('http')) {
    url = endpoint;
  } else {
    const baseUrl = API_BASE_URL.endsWith('/') 
      ? API_BASE_URL.slice(0, -1) 
      : API_BASE_URL;
      
    const cleanEndpoint = endpoint.startsWith('/') 
      ? endpoint.slice(1) 
      : endpoint;
      
    url = `${baseUrl}/${cleanEndpoint}`;
  }

  try {
    let response: Response;

    if (requiresAuth) {
      // Use the enhanced authenticatedFetch that handles token refresh automatically
      response = await authenticatedFetch(url, {
        ...fetchOptions,
        signal: fetchOptions.signal || (AbortSignal.timeout ? AbortSignal.timeout(20000) : undefined),
      });
    } else {
      // For non-auth requests, use regular fetch
      const headers = new Headers(fetchOptions.headers);
      if (!headers.has('Content-Type') && !(fetchOptions.body instanceof FormData)) {
        headers.set('Content-Type', 'application/json');
      }

      response = await fetch(url, {
        ...fetchOptions,
        headers,
        credentials: 'include',
        signal: fetchOptions.signal || (AbortSignal.timeout ? AbortSignal.timeout(20000) : undefined),
      });
    }

    if (!response.ok) {
      // Extract error message if possible
      let errorMessage = `API Error: ${response.status} ${response.statusText}`;
      
      try {
        const errorData = await response.json();
        errorMessage = errorData.message || errorData.error || errorMessage;
      } catch (_) {
        // If parsing JSON fails, stick with the default error message
      }
      
      throw new Error(errorMessage);
    }

    // Process successful response
    const contentType = response.headers.get('content-type');
    if (contentType && contentType.includes('application/json')) {
      const data = await response.json();
      
      // Store successful response in cache if applicable
      if (canUseCache) {
        apiCache.set(cacheKey, data);
      }
      
      return data as T;
    } else {
      return {} as T;
    }
  } catch (error: any) {
    // Handle and format the error using our enhanced error handler
    if (error.name === 'AbortError') {
      handleApiError(error, { 
        endpoint,
        fallbackMessage: 'Request timed out. Please try again.',
        showToast: true
      });
    } else if (error?.message?.includes('Session expired')) {
      // Session expired errors are already handled by authenticatedFetch
      handleApiError(error, {
        endpoint,
        showToast: true,
        fallbackMessage: 'Your session has expired. Please log in again.'
      });
    } else {
      // Other errors
      handleApiError(error, {
        endpoint,
        showToast: true,
        fallbackMessage: `Request to ${endpoint} failed`
      });
    }
    
    // Re-throw to maintain error flow
    throw error;
  }
}

/**
 * HTTP Methods wrappers for cleaner API calls
 */
const requestHandler = {
  /**
   * GET request with caching support
   */
  get: <T = any>(endpoint: string, options: FetchOptions = {}): Promise<T> => {
    return apiRequest<T>(endpoint, { ...options, method: 'GET', useCache: options.useCache !== false });
  },

  /**
   * POST request (invalidates cache for the endpoint)
   */
  post: <T = any>(endpoint: string, data?: any, options: FetchOptions = {}): Promise<T> => {
    // Clear cache for this endpoint on POST
    const baseEndpoint = endpoint.split('?')[0];
    apiCache.clear(baseEndpoint);
    
    return apiRequest<T>(endpoint, {
      ...options,
      method: 'POST',
      body: data ? JSON.stringify(data) : undefined,
      useCache: false,
    });
  },

  /**
   * PUT request (invalidates cache for the endpoint)
   */
  put: <T = any>(endpoint: string, data?: any, options: FetchOptions = {}): Promise<T> => {
    // Clear cache for this endpoint on PUT
    const baseEndpoint = endpoint.split('?')[0];
    apiCache.clear(baseEndpoint);
    
    return apiRequest<T>(endpoint, {
      ...options,
      method: 'PUT',
      body: data ? JSON.stringify(data) : undefined,
      useCache: false,
    });
  },

  /**
   * PATCH request (invalidates cache for the endpoint)
   */
  patch: <T = any>(endpoint: string, data?: any, options: FetchOptions = {}): Promise<T> => {
    // Clear cache for this endpoint on PATCH
    const baseEndpoint = endpoint.split('?')[0];
    apiCache.clear(baseEndpoint);
    
    return apiRequest<T>(endpoint, {
      ...options,
      method: 'PATCH',
      body: data ? JSON.stringify(data) : undefined,
      useCache: false,
    });
  },

  /**
   * DELETE request (invalidates cache for the endpoint)
   */
  delete: <T = any>(endpoint: string, options: FetchOptions = {}): Promise<T> => {
    // Clear cache for this endpoint on DELETE
    const baseEndpoint = endpoint.split('?')[0];
    apiCache.clear(baseEndpoint);
    
    return apiRequest<T>(endpoint, { 
      ...options, 
      method: 'DELETE',
      useCache: false,
    });
  },

  /**
   * Upload form data (files)
   */
  upload: <T = any>(endpoint: string, formData: FormData, options: FetchOptions = {}): Promise<T> => {
    // Clear cache for this endpoint on upload
    const baseEndpoint = endpoint.split('?')[0];
    apiCache.clear(baseEndpoint);
    
    return apiRequest<T>(endpoint, {
      ...options,
      method: 'POST',
      body: formData,
      useCache: false,
      // Don't set Content-Type header; browser will set it with boundary
      headers: { ...options.headers },
    });
  },
  
  /**
   * Clear cache for specific endpoint
   */
  clearCache: (endpoint: string): void => {
    apiCache.clear(endpoint);
  },
  
  /**
   * Clear all cache entries
   */
  clearAllCache: (): void => {
    apiCache.clearAll();
  }
};
